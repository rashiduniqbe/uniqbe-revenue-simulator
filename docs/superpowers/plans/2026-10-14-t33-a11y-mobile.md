## Review Focus

1. **Empty state.** With no product or no price there is no result, so the verdict slot must not render at all. Otherwise a sticky strip or a phantom grid row appears above the empty-state copy. Pinned in Task 1 by `no verdict slot until there is a result`.
2. **Tallest sticky stack.** The "marginal" verdict card has an extra explanatory line, so it is taller than the other two. `scroll-padding-top` must clear the tallest one or focused inputs hide behind it. Pinned in Task 2 by running the not-obscured test in the marginal state.
3. **320px reflow.** The header row (title + two market pills, now 44px tall) must wrap, not overflow. Pinned in Task 6 by `reflows at 320px`.
4. **Desktop layout must not regress.** The verdict moves to the top of the right column, above the threshold banner and warnings. That matches the spec §9.2 diagram. Pinned in Task 1 by `desktop: verdict sits in the right column and is not sticky`.
5. **Virtual keyboard on a small phone.** On an iPhone SE (667px tall) an open keyboard plus a sticky verdict card plus the sticky disclaimer may leave very little room for the field being edited. This cannot be asserted in Chromium. It is item 4 on the Task 7 checklist, with a stated fallback if it fails.

import AxeBuilder from "@axe-core/playwright";
import type { Page } from "@playwright/test";
import { expect } from "./fixtures";

export const VERDICT = /PROFITABLE|MARGINAL|LOSS-MAKING/;
export const WCAG_AA = ["wcag2a", "wcag2aa", "wcag21a", "wcag21aa"];

export async function expectNoViolations(page: Page): Promise<void> {
  const results = await new AxeBuilder({ page }).withTags(WCAG_AA).analyze();
  expect(results.violations.map((v) => `${v.id}: ${v.help} (${v.nodes.length})`)).toEqual([]);
}

export async function tabUntil(
  page: Page,
  predicate: string,
  key = "Tab",
  maxPresses = 40,
): Promise<void> {
  for (let i = 0; i < maxPresses; i++) {
    await page.keyboard.press(key);
    if (await page.evaluate(predicate)) return;
  }
  throw new Error(`focus never reached: ${predicate}`);
}

// 5% is below the verdict's marginal threshold (VERDICT_MARGIN_THRESHOLD_PCT), so the applied
// price lands on a MARGINAL verdict for any product, with no price hardcoded in the test.
export async function reachMarginalVerdict(page: Page): Promise<void> {
  await page.getByLabel("Suggest a price for a target margin").fill("5");
  await page.getByRole("button", { name: "Suggest", exact: true }).click();
  await page.getByRole("button", { name: "Apply", exact: true }).click();
  await expect(page.getByText("MARGINAL")).toBeVisible();
}

import type { Page } from "@playwright/test";
import { expect, test } from "./fixtures";
import { reachMarginalVerdict } from "./a11y-helpers";
import { firstCatalogueItem, pickFirstProduct } from "./helpers";

const MIN_RING_PX = 2;
const MAX_TAB_STOPS = 80;

interface Stop {
  name: string;
  ringStyle: string;
  ringPx: number;
  overlapsSlot: boolean;
  overlapsNote: boolean;
}

// Presses Tab through the page and describes every focus stop until focus leaves the document.
async function tabStops(page: Page): Promise<Stop[]> {
  const stops: Stop[] = [];
  for (let i = 0; i < MAX_TAB_STOPS; i++) {
    await page.keyboard.press("Tab");
    const stop = await page.evaluate(() => {
      const el = document.activeElement;
      if (!el || el === document.body) return null;
      const style = getComputedStyle(el);
      const rect = el.getBoundingClientRect();
      const overlaps = (selector: string): boolean => {
        const other = document.querySelector(selector)?.getBoundingClientRect();
        return !!other && rect.bottom > other.top && rect.top < other.bottom;
      };
      return {
        name: el.id || `${el.tagName.toLowerCase()}: ${(el.textContent ?? "").trim().slice(0, 24)}`,
        ringStyle: style.outlineStyle,
        ringPx: parseFloat(style.outlineWidth),
        overlapsSlot: overlaps('[data-testid="verdict-slot"]'),
        overlapsNote: overlaps('[role="note"]'),
      };
    });
    if (stop === null) break;
    stops.push(stop);
  }
  return stops;
}

test("every keyboard focus stop shows a ring of at least 2px", async ({ page }) => {
  const item = firstCatalogueItem();
  await page.goto(`/?p=${item.code}&sp=9999.00`);
  const stops = await tabStops(page);
  expect(stops.length, "the page has focusable controls").toBeGreaterThan(5);
  const weak = stops
    .filter((s) => s.ringStyle === "none" || s.ringPx < MIN_RING_PX)
    .map((s) => `${s.name} (${s.ringStyle} ${s.ringPx}px)`);
  expect(weak).toEqual([]);
});

test("mobile: reflows at 320px with no horizontal scroll", async ({ page }, testInfo) => {
  test.skip(testInfo.project.name !== "mobile-portrait", "mobile-only check");
  await page.setViewportSize({ width: REFLOW_WIDTH, height: REFLOW_HEIGHT });
  const item = firstCatalogueItem();
  const urls = [
    `/?p=${item.code}&sp=9999.00`,
    `/?m=AU&pl=shopify&p=${item.code}&sp=9999.00`,
    `/compare?p=${item.code}&sp_uk=9999.00&sp_au=19999.00`,
  ];
  for (const url of urls) {
    await page.goto(url);
    await expect(page.getByText(VERDICT).first()).toBeVisible();
    const report = await page.evaluate((viewportWidth) => {
      const offenders = [...document.querySelectorAll("body *")]
        .map((el) => ({ el, right: el.getBoundingClientRect().right }))
        .filter(({ right }) => right > viewportWidth + 1)
        .sort((a, b) => b.right - a.right);
      const first = offenders[0];
      return {
        overflow: document.documentElement.scrollWidth - viewportWidth,
        offender: first ? `${first.el.tagName.toLowerCase()}.${first.el.className}` : null,
      };
    }, REFLOW_WIDTH);
    expect(report.offender, `${url}: widest element past the right edge`).toBeNull();
    expect(report.overflow, `${url} overflows horizontally`).toBeLessThanOrEqual(0);
  }
});
